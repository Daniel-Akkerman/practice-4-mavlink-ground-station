cmake_minimum_required(VERSION 3.21)
project(Practical_04_MavlinkGroundStation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Network Charts Test)
add_library(station mavlinkmessageparser.cpp telemetrymodel.cpp mavlinkmanager.cpp uavsimulator.cpp)
target_link_libraries(station PUBLIC Qt6::Network)
add_library(station_ui mainwindow.cpp mainwindow.ui)
target_link_libraries(station_ui PUBLIC station Qt6::Widgets Qt6::Charts)
add_executable(GroundStation main.cpp)
target_link_libraries(GroundStation PRIVATE station_ui)
add_executable(UavSimulator simulator_main.cpp)
target_link_libraries(UavSimulator PRIVATE station Qt6::Widgets)
enable_testing()
add_executable(StationTests tests.cpp)
target_link_libraries(StationTests PRIVATE station_ui Qt6::Test)
add_test(NAME station COMMAND StationTests -o station-tests.txt,txt)
set_tests_properties(station PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
